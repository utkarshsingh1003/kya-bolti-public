-- ============================
-- Row Level Security (RLS) policies
-- These policies are safe to run; they deny anon access and allow service_role full access.
-- The application uses custom session auth (AUTH_SECRET), so anon access is blocked.
-- If Supabase Auth is later enabled, adjust policies accordingly.
-- ============================
-- Enable RLS on critical tables
ALTER TABLE users ENABLE ROW LEVEL SECURITY;
ALTER TABLE ad_inquiries ENABLE ROW LEVEL SECURITY;
ALTER TABLE quiz_scores ENABLE ROW LEVEL SECURITY;
ALTER TABLE poll_requests ENABLE ROW LEVEL SECURITY;

-- Deny all access to anon users (anonymous browser sessions)
CREATE POLICY "users_no_anon" ON users FOR ALL TO anon USING (false) WITH CHECK (false);
CREATE POLICY "ad_inquiries_no_anon" ON ad_inquiries FOR ALL TO anon USING (false) WITH CHECK (false);
CREATE POLICY "quiz_scores_no_anon" ON quiz_scores FOR ALL TO anon USING (false) WITH CHECK (false);
CREATE POLICY "poll_requests_no_anon" ON poll_requests FOR ALL TO anon USING (false) WITH CHECK (false);

-- Allow the Supabase service_role (server‑only) full access to these tables
CREATE POLICY "service_role_full_access_users" ON users FOR ALL TO service_role USING (true) WITH CHECK (true);
CREATE POLICY "service_role_full_access_ad_inquiries" ON ad_inquiries FOR ALL TO service_role USING (true) WITH CHECK (true);
CREATE POLICY "service_role_full_access_quiz_scores" ON quiz_scores FOR ALL TO service_role USING (true) WITH CHECK (true);
CREATE POLICY "service_role_full_access_poll_requests" ON poll_requests FOR ALL TO service_role USING (true) WITH CHECK (true);

      expected.column_definition
    );
  end loop;

  if cardinality(missing) > 0 then
    raise warning 'KBP 004: skipped missing tables: %', array_to_string(missing, ', ');
  end if;
end $$;

========== db/migrations/005_rls_policies.sql ==========
-- Kya Bolti Public — migration 005
-- Supabase/PostgREST defense in depth.
--
-- The application talks to Postgres directly as the table owner, so RLS is not
-- part of its authorization model. These policies only close the anonymous
-- PostgREST surface. Rules:
--  * Do nothing when the Supabase roles are absent (plain Postgres, Neon, CI).
--  * Never enable RLS when the migration role is anon/authenticated, and skip a
--    table when the migration role neither owns it nor bypasses RLS. This keeps
--    login, voting, submissions and admin access working.
--  * Recognise the policy names created by 002 as well as the canonical names
--    used here, so re-running cannot create duplicate permissive policies.
--  * Protect schema_migrations from anon/authenticated access.

do $$
declare
  tbl text;
  actor text := current_user;
  actor_bypasses boolean := false;
  owns_table boolean := false;
  app_tables text[] := array[
    'polls',
    'poll_options',
    'poll_votes',
    'issues',
    'reviews',
    'threads',
    'replies',
    'questions',
    'reports',
    'representatives',
    'contact_messages',
    'users',
    'quiz_scores',
    'poll_requests',
    'announcements',
    'buzz_posts',
    'ad_inquiries',
    'news_sources',
    'moderation_actions'
  ];
begin
  if not exists (select 1 from pg_roles where rolname = 'anon')
     and not exists (select 1 from pg_roles where rolname = 'service_role') then
    raise notice 'KBP 005: anon/service_role roles are absent; skipping Supabase RLS setup';
    return;
  end if;

  if actor in ('anon', 'authenticated') then
    raise exception 'KBP 005: current_user is %; refusing to enable RLS because that would lock out the application', actor
      using hint = 'Run migrations as the table owner (for example postgres) or service_role.';
  end if;

  select coalesce(bool_or(rolsuper or rolbypassrls), false)
  into actor_bypasses
  from pg_roles
  where rolname = actor;

  foreach tbl in array app_tables loop
    if to_regclass(format('public.%I', tbl)) is null then
      raise notice 'KBP 005: skipping missing table %', tbl;
      continue;
    end if;

    select exists (
      select 1
      from pg_class c
      join pg_roles r on r.oid = c.relowner
      where c.oid = to_regclass(format('public.%I', tbl))
        and r.rolname = actor
    ) into owns_table;

    if not owns_table and not actor_bypasses then
      raise warning 'KBP 005: current_user % neither owns % nor bypasses RLS; skipping to preserve access', actor, tbl;
      continue;
    end if;

    execute format('alter table public.%I enable row level security', tbl);

    -- Deny anon exactly once. 002 used <table>_no_anon; this migration uses
    -- <table>_anon_deny. Checking both names makes the operation idempotent.
    if exists (select 1 from pg_roles where rolname = 'anon')
       and not exists (
         select 1
         from pg_policies
         where schemaname = 'public'
           and tablename = tbl
           and policyname::text in (tbl || '_no_anon', tbl || '_anon_deny')
       ) then
      execute format(
        'create policy %I on public.%I for all to anon using (false) with check (false)',
        tbl || '_anon_deny',
        tbl
      );
    end if;

    -- Allow service_role exactly once. 002 used service_role_full_access_<table>;
    -- any existing permissive FOR ALL service_role policy is treated as a match.
    if exists (select 1 from pg_roles where rolname = 'service_role')
       and not exists (
         select 1
         from pg_policies
         where schemaname = 'public'
           and tablename = tbl
           and 'service_role'::name = any(roles)
           and cmd = 'ALL'
       ) then
      execute format(
        'create policy %I on public.%I for all to service_role using (true) with check (true)',
        tbl || '_service_all',
        tbl
      );
    end if;
  end loop;

  -- Migration history is infrastructure, not application data. Enable RLS (no
  -- policies means deny-by-default for non-bypassing roles) and revoke the
  -- default Supabase grants.
  if to_regclass('public.schema_migrations') is not null then
    select exists (
      select 1
      from pg_class c
      join pg_roles r on r.oid = c.relowner
      where c.oid = to_regclass('public.schema_migrations')
        and r.rolname = actor
    ) into owns_table;

    if owns_table or actor_bypasses then
      execute 'alter table public.schema_migrations enable row level security';

      if exists (select 1 from pg_roles where rolname = 'anon') then
        execute 'revoke all on table public.schema_migrations from anon';
      end if;

      if exists (select 1 from pg_roles where rolname = 'authenticated') then
        execute 'revoke all on table public.schema_migrations from authenticated';
      end if;
    else
      raise warning 'KBP 005: skipping schema_migrations protection because current_user % does not own it', actor;
    end if;
  end if;
end $$;

========== db/migrations/006_foreign_keys_and_indexes.sql ==========
-- Kya Bolti Public — migration 006
-- Ownership foreign keys only.
--
-- Rules:
--  * Only the four owner_id -> users(id) keys are added. The composite
--    poll_votes(option_id, poll_id) key is deliberately omitted: castVote()
--    already verifies option ownership, so the extra key/index is redundant.
--  * FKs are added NOT VALID to avoid a long table scan and lock. They are
--    validated in a second pass only when the orphan count is zero.
--  * Missing tables, missing columns and incompatible types are skipped with a
--    warning instead of aborting the migration.
--  * No redundant indexes are added.

do $$
declare
  rec record;
  orphans bigint;
  users_oid oid := to_regclass('public.users');
  users_id_type text;
  users_id_attnum smallint;
begin
  if users_oid is null then
    raise notice 'KBP 006: public.users is missing; skipping owner foreign keys';
    return;
  end if;

  select a.attnum
  into users_id_attnum
  from pg_attribute a
  where a.attrelid = users_oid
    and a.attname = 'id'
    and not a.attisdropped;

  select c.data_type
  into users_id_type
  from information_schema.columns c
  where c.table_schema = 'public'
    and c.table_name = 'users'
    and c.column_name = 'id';

  if users_id_attnum is null
     or users_id_type not in ('text', 'character varying') then
    raise warning 'KBP 006: public.users.id is missing or not a text type; skipping owner foreign keys';
    return;
  end if;

  -- A foreign key needs a primary key or unique constraint on users(id).
  if not exists (
    select 1
    from pg_constraint c
    where c.conrelid = users_oid
      and c.contype in ('p', 'u')
      and c.conkey = array[users_id_attnum]
  ) then
    raise warning 'KBP 006: public.users.id has no primary key or unique constraint; skipping owner foreign keys';
    return;
  end if;

  -- Pass 1: add the keys as NOT VALID. Existing rows are not scanned.
  for rec in
    select *
    from (values
      ('issues', 'issues_owner_id_fkey'),
      ('reviews', 'reviews_owner_id_fkey'),
      ('threads', 'threads_owner_id_fkey'),
      ('questions', 'questions_owner_id_fkey')
    ) as owner_fks(table_name, constraint_name)
  loop
    if to_regclass(format('public.%I', rec.table_name)) is null then
      raise notice 'KBP 006: skipping % because %. is missing', rec.constraint_name, rec.table_name;
      continue;
    end if;

    if not exists (
      select 1
      from information_schema.columns c
      where c.table_schema = 'public'
        and c.table_name = rec.table_name
        and c.column_name = 'owner_id'
    ) then
      raise notice 'KBP 006: skipping % because %.owner_id is missing', rec.constraint_name, rec.table_name;
      continue;
    end if;

    if not exists (
      select 1
      from information_schema.columns c
      where c.table_schema = 'public'
        and c.table_name = rec.table_name
        and c.column_name = 'owner_id'
        and c.data_type in ('text', 'character varying')
    ) then
      raise notice 'KBP 006: skipping % because %.owner_id is not a text type', rec.constraint_name, rec.table_name;
      continue;
    end if;

    if exists (
      select 1
      from pg_constraint c
      where c.conname::text = rec.constraint_name
        and c.conrelid = to_regclass(format('public.%I', rec.table_name))
    ) then
      continue;
    end if;

    -- Skip if an equivalent owner_id -> users(id) key already exists under a
    -- different name.
    if exists (
      select 1
      from pg_constraint c
      where c.conrelid = to_regclass(format('public.%I', rec.table_name))
        and c.contype = 'f'
        and c.confrelid = users_oid
        and exists (
          select 1
          from pg_attribute a
          where a.attrelid = c.conrelid
            and a.attnum = any(c.conkey)
            and a.attname = 'owner_id'
        )
    ) then
      continue;
    end if;

    execute format(
      'alter table public.%I add constraint %I foreign key (owner_id) references public.users(id) on delete set null not valid',
      rec.table_name,
      rec.constraint_name
    );
  end loop;

  -- Pass 2: validate only when there are no orphaned owner_id values.
  for rec in
    select *
    from (values
      ('issues', 'issues_owner_id_fkey'),
      ('reviews', 'reviews_owner_id_fkey'),
      ('threads', 'threads_owner_id_fkey'),
      ('questions', 'questions_owner_id_fkey')
    ) as owner_fks(table_name, constraint_name)
  loop
    if to_regclass(format('public.%I', rec.table_name)) is null then
      continue;
    end if;

    if not exists (
      select 1
      from information_schema.columns c
      where c.table_schema = 'public'
        and c.table_name = rec.table_name
        and c.column_name = 'owner_id'
    ) then
      continue;
    end if;

    if not exists (
      select 1
      from pg_constraint c
      where c.conname::text = rec.constraint_name
        and c.conrelid = to_regclass(format('public.%I', rec.table_name))
    ) then
      continue;
    end if;

    if exists (
      select 1
      from pg_constraint c
      where c.conname::text = rec.constraint_name
        and c.conrelid = to_regclass(format('public.%I', rec.table_name))
        and c.convalidated
    ) then
      continue;
    end if;

    execute format(
      'select count(*) from public.%I t where t.owner_id is not null and not exists (select 1 from public.users u where u.id = t.owner_id)',
      rec.table_name
    ) into orphans;

    if orphans > 0 then
      raise warning 'KBP 006: % orphaned owner_id row(s) in %; constraint % left NOT VALID', orphans, rec.table_name, rec.constraint_name;
    else
      execute format(
        'alter table public.%I validate constraint %I',
        rec.table_name,
        rec.constraint_name
      );
    end if;
  end loop;
end $$;
